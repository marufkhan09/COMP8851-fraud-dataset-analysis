            if self.add_self:
                rst = h_self + h_neigh
            else:
                rst = h_neigh
            # rst = torch.concat([rst,graph.dstdata['degree']],-1)
            # bias term
            if self.bias is not None:
                rst = rst + self.bias
            # activation
            if self.activation is not None:
                rst = self.activation(rst)
            # normalization
            if self.norm is not None:
                rst = self.norm(rst)
            return rst  # , h_self
